# Flycode 前端功能开发任务

## 项目背景

Flycode 是一个共创网站，当前已有完整的前端界面。你需要为三个新功能开发前端交互界面。

**技术栈：**
- 纯原生 JavaScript（不使用框架）
- 主要文件：`public/app.js`（已有完整代码）
- 样式文件：`public/styles.css`
- HTML 结构：`public/index.html`

**当前状态：**
- 已有提案提交、投票、管理后台等功能
- 使用 `X-Visitor-Id` 请求头标识访客
- 已有的访客 ID 生成：`localStorage.getItem('flycode-visitor-id')` 或自动生成

async function handleLikeClick(event) {
  const btn = event.currentTarget;
  const proposalId = btn.dataset.proposalId;
  
  try {
    btn.disabled = true;
    
    const response = await fetch('/api/likes', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Visitor-Id': getVisitorId()
      },
      body: JSON.stringify({ proposalId })
    });
    
    const data = await response.json();
    
    if (data.ok) {
      // 更新按钮状态
      if (data.action === 'liked') {
        btn.classList.add('liked');
      } else {
        btn.classList.remove('liked');
      }
      
      // 更新点赞数
      const countSpan = btn.querySelector('.like-count');
      countSpan.textContent = data.likeCount;
      
      // 可选：添加动画效果
      btn.style.transform = 'scale(1.1)';
      setTimeout(() => {
        btn.style.transform = 'scale(1)';
      }, 200);
    } else {
      showNotification(data.error || '点赞失败', 'error');
    }
  } catch (err) {
    console.error('点赞失败:', err);
    showNotification('网络错误，请稍后重试', 'error');
  } finally {
    btn.disabled = false;
  }
}

**2. 加载评论列表：**

```javascript
async function loadComments(proposalId) {
  const container = document.getElementById('comments-list');
  container.innerHTML = '<p class="loading-text">加载中...</p>';
  
  try {
    const response = await fetch(`/api/comments?proposalId=${proposalId}`);
    const data = await response.json();
    
    if (data.ok) {
      if (data.comments.length === 0) {
        container.innerHTML = '<p class="empty-state">还没有评论，来抢沙发吧！</p>';
      } else {
        container.innerHTML = data.comments.map(comment => `
          <div class="comment-item">
            <div class="comment-header">
              <span class="comment-author">${escapeHtml(comment.author)}</span>
              <span class="comment-time">${formatDate(comment.createdAt)}</span>
            </div>
            <div class="comment-content">${escapeHtml(comment.content)}</div>
          </div>
        `).join('');
      }
    } else {
      container.innerHTML = '<p class="empty-state">加载失败，请刷新重试</p>';
    }
  } catch (err) {
    console.error('加载评论失败:', err);
    container.innerHTML = '<p class="empty-state">加载失败，请刷新重试</p>';
  }
}

let allProposals = []; // 缓存所有提案
let currentFilterTag = ''; // 当前筛选标签

async function loadTagFilters() {
  try {
    const response = await fetch('/api/proposals/tags');
    const data = await response.json();
    
    if (data.ok) {
      const container = document.getElementById('tag-filter-bar');
      
      // 渲染标签按钮
      container.innerHTML = `
        <button class="filter-tag active" data-tag="">
          全部
        </button>
        ${data.tags.map(tag => `
          <button class="filter-tag" data-tag="${escapeHtml(tag.name)}">
            ${escapeHtml(tag.name)}
            <span class="tag-count">(${tag.count})</span>
          </button>
        `).join('')}
      `;
      
      // 绑定点击事件
      container.querySelectorAll('.filter-tag').forEach(btn => {
        btn.addEventListener('click', handleTagFilter);
      });
    }
  } catch (err) {
    console.error('加载标签失败:', err);
  }
}

// 标签筛选处理
function handleTagFilter(event) {
  const btn = event.currentTarget;
  const tag = btn.dataset.tag;
  
  // 更新激活状态
  document.querySelectorAll('.filter-tag').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  
  // 保存当前筛选
  currentFilterTag = tag;
  
  // 筛选提案
  filterProposals(tag);
}

function filterProposals(tag) {
  if (!tag) {
    // 显示全部
    renderProposals(allProposals);
  } else {
    // 只显示包含该标签的提案
    const filtered = allProposals.filter(p => 
      p.tags && p.tags.includes(tag)
    );
    renderProposals(filtered);
  }
}
```
